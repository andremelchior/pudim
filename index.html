namespace Sobremesas.Brasil;

public abstract class Sobremesa
{
    public abstract string Nome { get; }
    public virtual void Servir()
    {
        Console.WriteLine("Servindo sobremesa tradicional.");
    }
}

public class Pudim : Sobremesa
{
    public override string Nome => "Pudim de Leite Condensado";
    public string Textura { get; set; } = "Aveludada & Cremosa";
    public bool TemFurinhos { get; set; } = true;
    public double NivelCaldaDeCaramelo { get; set; } = 100.0;

    public override void Servir()
    {
        Console.WriteLine("Servindo uma fatia aveludada com calda roxa.");
    }

    public async Task JiggleAsync()
    {
        Console.WriteLine("Wooble wooble!");
        await Task.Yield();
    }
}
